$dnsServer = "BLPAZCUSDC03.capgeminibrim.com"

# Define the zone name
$zoneName = "capgeminibrim.com"

# Specify the path to the CSV file containing A records and IP addresses
$csvPath = "C:\temp\dns1.csv"

# Import the CSV file
$aRecordsToDelete = Import-Csv $csvPath

# Iterate through each row in the CSV

    $recordName = $record.Name
    $recordFqdn = $recordName + "." + $zoneName

    # Check if the A record exists with the specified hostname and IP address
    $existingRecord = Get-DnsServerResourceRecord -ZoneName $zoneName -Name $recordName -RRType A -ErrorAction SilentlyContinue 

    if ($existingRecord) {
